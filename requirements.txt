# FreeProxyPool 运行依赖（v0.1.0）
#
# 版本约束原则：
#   1. 全部使用 `>=下限,<上限` 区间。旧版只写 `>=`，意味着
#      任何一次 pip 解析都可能拉到不兼容的大版本，是线上事故的常见来源。
#   2. 上限取"下一个已知破坏性变更之前"的最后一个版本，由自检套件验证。
#   3. 已移除的依赖及原因见 docs/05-功能优化与新增日志.md：
#        - Flask / gunicorn  v0.1.0 删除了整套 Flask 实现，只保留 FastAPI
#        - lxml             全项目未使用（采集走正则，不解析 HTML DOM）
#        - supervisor       仅 Docker 镜像内使用，改为 apt 安装

# ---------- Web 框架 ----------
fastapi>=0.115,<1.0
uvicorn[standard]>=0.30,<1.0
# starlette 下限 1.3.1 是**安全底线**，不是功能需求：
# 0.41~1.3.0 之间存在已公开漏洞（pip-audit 实测，2026-10-13）：
#   PYSEC-2026-161  Host 头未校验即用于重建 URL（需 1.0.1+）
#   PYSEC-2026-2280 HTTPEndpoint 按方法名选处理器（需 1.1.0+）
#   PYSEC-2026-2281 Windows 下 StaticFiles 路径解析（需 1.1.0+）
#   PYSEC-2026-248  请求路径未校验即用于重建（需 1.3.0+）
#   PYSEC-2026-249  request.form() 无 max_fields/max_part_size 上限（需 1.3.1+）
# 旧约束写的是 >=0.41,<1.0，恰好把全部修复版挡在外面 —— 开发机碰巧装了
# 1.6.0 所以自检全绿，而 Docker 镜像装的是 0.52.1，一直带着漏洞运行。
starlette>=1.3.1,<2.0
pydantic>=2.9,<3.0

# ---------- 存储 ----------
redis>=5.0,<6.0          # 5.x 起 decode_responses / RESP2 行为稳定

# ---------- 异步校验 ----------
aiohttp>=3.10,<4.0

# ---------- SOCKS 代理支持（v0.1.0 P0-1） ----------
# aiohttp 的 `session.get(proxy=...)` 只能逐请求切换 HTTP 代理；
# SOCKS 代理需要用连接器在**构造时**绑定代理地址
# （aiohttp_socks.ProxyConnector），因此每个 SOCKS 代理一个会话。
# 0.12.0 为当前实测版本；上限取 <1.0（尚无 1.x，不存在已知破坏性变更）。
aiohttp-socks>=0.10,<1.0

# ---------- 采集 HTTP ----------
requests>=2.32,<3.0

# ---------- MySQL 冷存储归档（可选） ----------
SQLAlchemy>=2.0,<3.0
PyMySQL>=1.1,<2.0

# ---------- 调度 ----------
APScheduler>=3.10,<4.0

# ---------- 监控 ----------
prometheus-client>=0.20,<1.0

# ---------- CLI ----------
click>=8.1,<9.0

# ---------- 国际化时区（容器内 TZ 依赖系统 tzdata，此处仅为校验目标时区） ----------
tzdata>=2024.1

# ---------- GeoIP / ASN（v0.1.0 第三批优化） ----------
maxminddb>=1.5,<2.0          # GeoLite2 Country + ASN mmdb 读取器（纯 Python reader）
